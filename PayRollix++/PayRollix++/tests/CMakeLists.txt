cmake_minimum_required(VERSION 3.20)
project(PayrollixCoreTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(core_tests CoreTests.cpp ../src/Store.cpp ../src/PayrollCalculator.cpp)
target_include_directories(core_tests PRIVATE ../include)
if(WIN32)
    target_sources(core_tests PRIVATE ../src/Security.cpp)
    target_link_libraries(core_tests PRIVATE winsqlite3 bcrypt)
    target_compile_definitions(core_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    find_package(SQLite3 REQUIRED)
    target_sources(core_tests PRIVATE SecurityTestDouble.cpp)
    target_link_libraries(core_tests PRIVATE SQLite::SQLite3)
endif()
add_test(NAME payroll_core COMMAND core_tests)
add_executable(windows_version_tests WindowsVersionTests.cpp)
target_include_directories(windows_version_tests PRIVATE ../include)
add_test(NAME windows_version_policy COMMAND windows_version_tests)

# Syntax/object check only: does not produce a runnable non-Windows application.
option(PAYROLLIX_CHECK_QT_UI "Compile Qt UI objects without linking the Windows app" OFF)
if(PAYROLLIX_CHECK_QT_UI)
    find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets)
    add_library(qt_ui_check OBJECT ../src/AppWindow.cpp ../src/EmployeeViews.cpp
        ../src/AdminViews.cpp ../src/ReportViews.cpp)
    target_include_directories(qt_ui_check PRIVATE ../include)
    target_link_libraries(qt_ui_check PRIVATE Qt6::Widgets)
endif()
